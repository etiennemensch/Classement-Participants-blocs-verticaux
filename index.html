function parseExcelText(rawText) { const extracted = []; const lines = rawText.split(/\r?\n/); lines.forEach((line) => { if (!line.trim()) return; // Découpe par tabulation (copie d'Excel) ou par espaces multiples let tokens = line.split('\t').map(t => t.trim()).filter(t => t.length > 0); if (tokens.length < 3) { tokens = line.trim().split(/\s{2,}/); } if (tokens.length >= 3) { // La première colonne est la place / Rang (Pl) const rank = parseInt(tokens[0], 10); if (!isNaN(rank) && rank > 0 && rank < 1000) { let title = ''; let name = ''; let elo = '2000'; let fed = 'FRA'; let pts = ''; let tr = ''; const validTitles = ['g', 'm', 'f', 'gf', 'mf', 'ff', 'gmi', 'mi', 'fm']; // On parcourt les colonnes après le rang tokens.slice(1).forEach((token) => { const lower = token.toLowerCase(); // Titre FIDE (g, m, f...) if (validTitles.includes(lower)) { title = token.toUpperCase(); } // Fédération FIDE 3 lettres (FRA, IND, UKR, etc.) else if (token.match(/^[A-Z]{3}$/) && token !== 'FFE' && token !== 'FID') { fed = token; } // Elo (4 chiffres, ex: 2624 F, 2528) else if (token.match(/^\d{3,4}\s*[FfNn]?$/)) { elo = token.replace(/[^\d]/g, ''); } // Points / Tronqué (ex: 4, 3½, 3.5, 6, 6½) - Se détache du numéro de rang else if (token.match(/^(\d+(\.5|½|\/2|\,\d)?)$/)) { // On s'assure qu'on ne reprend pas la valeur du rang if (parseInt(token, 10) !== rank || token.includes('½') || token.includes('.')) { if (!pts) { pts = token; } else if (!tr) { tr = token; } } } // Nom du joueur else if (token.match(/[A-Za-zÀ-ÿ]{2,}/) && !token.match(/^\d+$/)) { if (!name) name = token; else name += ' ' + token; } }); if (name) { extracted.push({ rank: rank, title: title, name: name, elo: elo, fed: fed, pts: pts || '0', tr: tr || '' }); } } } }); if (extracted.length > 0) { extracted.sort((a, b) => a.rank - b.rank); players = extracted; document.getElementById('statusText').textContent = `🟢 ${players.length} joueurs importés`; applySettings(); } else { alert("Impossible de lire les données. Assurez-vous de sélectionner toutes les cellules dans Excel (Ctrl+A) et de les coller."); } }
